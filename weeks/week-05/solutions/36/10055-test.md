# 核心邏輯：讀入兩數，輸出絕對差

import sys

def solve():
    # 使用 sys.stdin 讀取每一行直到檔案結束 (EOF)
    for line in sys.stdin:
        # 將每一行拆分為字串列表，並轉為整數
        # 使用 map(int, ...) 一次轉換多個數字
        try:
            nums = list(map(int, line.split()))
            if len(nums) == 2:
                a, b = nums
                # 輸出絕對值：無論誰大誰小，答案都一樣
                print(abs(a - b))
        except EOFError:
            break
        except ValueError:
            # 預防空白行導致的轉換錯誤
            continue

if __name__ == "__main__":
    solve()
