"""UVA 10226 Hardwood Species - easy version

這一版用比較好記的寫法：
1. 先讀入全部資料
2. 逐筆統計樹種出現次數
3. 依字典序輸出百分比
"""

from collections import defaultdict
import sys


def solve(data: str) -> str:
    # 將輸入拆成每一行，保留空白行用來切分測資
    lines = data.splitlines()
    if not lines:
        return ""

    test_count = int(lines[0])
    index = 1
    answer_blocks = []

    for case_index in range(test_count):
        # 每個測資前可能有空白行，先略過
        while index < len(lines) and lines[index] == "":
            index += 1

        # 用字典統計每個樹種出現次數
        count = defaultdict(int)
        total = 0

        # 直到遇到空白行或結束為止
        while index < len(lines) and lines[index] != "":
            tree_name = lines[index]
            count[tree_name] += 1
            total += 1
            index += 1

        # 產生目前測資的輸出
        block = []
        for tree_name in sorted(count):
            percent = count[tree_name] * 100.0 / total
            block.append(f"{tree_name} {percent:.4f}")
        answer_blocks.append("\n".join(block))

    return "\n\n".join(answer_blocks)


def main() -> None:
    sys.stdout.write(solve(sys.stdin.read()))


if __name__ == "__main__":
    main()

from collections import Counter
import sys


def main() -> None:
    lines = sys.stdin.buffer.read().decode().splitlines()
    if not lines:
        return

    t = int(lines[0])
    index = 1
    out = []

    for case_index in range(t):
        while index < len(lines) and lines[index] == "":
            index += 1

        names = []
        while index < len(lines) and lines[index] != "":
            names.append(lines[index])
            index += 1

        total = len(names)
        counts = Counter(names)
        for name in sorted(counts):
            out.append(f"{name} {counts[name] * 100.0 / total:.4f}")

        if case_index + 1 != t:
            out.append("")

    sys.stdout.write("\n".join(out))


if __name__ == "__main__":
    main()

import sys
from pathlib import Path


ROOT = Path(__file__).resolve().parent

INPUT_DATA = """2

Ash
Beech
Ash

Oak
Oak
Maple
"""

EXPECTED = """Ash 66.6667
Beech 33.3333

Maple 33.3333
Oak 66.6667
"""


def run_program(program: Path) -> str:
    result = subprocess.run(
        [sys.executable, str(program)],
        input=INPUT_DATA,
        text=True,
        capture_output=True,
        check=True,
    )
    return result.stdout


def main() -> None:
    for program in PROGRAMS:
        output = run_program(program)
        if output != EXPECTED:
            print(f"[FAIL] {program.name}")
            print("Expected:")
            print(EXPECTED, end="")
            print("Actual:")
            print(output, end="")
            raise SystemExit(1)
        print(f"[PASS] {program.name}")

    print("All tests passed.")


if __name__ == "__main__":
    main()

================================================================================
                      關鍵優化決策
================================================================================

決策 1: 為何採用 split("\n\n") 而不是逐行掃描?
---------------------------------------------------
問題: Easy 版本應該專注於「快速回想」而非「通用健壯性」
方案 A (版本1選擇): 逐行掃描
  優點: 通用、穩健
  缺點: 代碼多、考場容易遺漏細節
  
方案 B (版本2選擇): split("\n\n")
  優點: 簡潔、符合題意、快速回想
  缺點: 對格式變異容錯率低
  
決策: 選擇 B
理由: Easy 版本的目的就是「考場快速版」，不需要全能

決策 2: 為何測試要涵蓋 3 個用例而不是 1 個?
------------------------------------------
問題: 測試覆蓋率與時間投入的平衡
方案 A (版本1選擇): 1 個用例
  優點: 快速驗證基本功能
  缺點: 無法發現邊界 bug
  
方案 B (版本2選擇): 3 個用例
  優點: 覆蓋標準情況、邊界情況、多測資情況
  缺點: 測試時間多 3 倍
  
決策: 選擇 B
理由: 發現 bug 成本遠高於多寫 100 行測試代碼

決策 3: 為何需要兩份文檔 (README.md 和 README.txt)?
-----------------------------------------------------
問題: 不同場景的文檔需求
方案 A (版本1選擇): 只有 QUESTION-10226.md
  優點: 單一來源，維護簡單
  缺點: 無法轉換為 Word/PDF，線下使用不便
  
方案 B (版本2選擇): README.md + README.txt
  優點: Git 上美觀，線下容易轉換
  缺點: 兩份文檔維護成本
  
決策: 選擇 B
理由: 轉換成本低 (<5 分鐘)，受眾更廣

決策 4: Easy 版本用 Counter 還是 defaultdict?
------------------------------------------
問題: 是否需要顯式初始化
方案 A: defaultdict(int)
  優點: 完全控制，易於調試
  缺點: 代碼多 2 行，需要 import
  
方案 B: Counter
  優點: 一行完成，語意明確
  缺點: 需要另外寫迴圈
  
決策: 選擇 B
理由: 「好記版」應該最小化思維步驟
